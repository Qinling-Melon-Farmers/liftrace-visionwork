cmake_minimum_required(VERSION 3.0.2)
project(uav_high_view)
find_package(catkin REQUIRED)
catkin_python_setup()
catkin_package()
catkin_install_python(PROGRAMS scripts/high_view_replay.py scripts/high_view_evaluate.py scripts/high_view_gazebo_capture.py scripts/high_view_probe_gate.py scripts/high_view_status_recorder.py DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(DIRECTORY config launch DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
if(CATKIN_ENABLE_TESTING)
  catkin_add_nosetests(test/test_core.py)
endif()
