cmake_minimum_required(VERSION 3.0.2)
project(uav_coverage_memory)
find_package(catkin REQUIRED COMPONENTS rospy sensor_msgs nav_msgs std_msgs std_srvs tf2_ros cv_bridge)
catkin_python_setup()
catkin_package()
catkin_install_python(PROGRAMS scripts/coverage_memory_shadow.py DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(DIRECTORY launch config DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
if(CATKIN_ENABLE_TESTING)
  catkin_add_nosetests(test/test_memory.py)
  catkin_add_nosetests(test/test_shadow_contract.py)
  catkin_add_nosetests(test/test_recording.py)
  catkin_add_nosetests(test/test_cloud.py)
  catkin_add_nosetests(test/test_queries.py)
endif()
